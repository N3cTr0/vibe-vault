# tools\sync-vault.ps1

```powershell
<#
.SYNOPSIS
  Regenerate the Obsidian vault's Octavia code snapshot from this repo.

.DESCRIPTION
  Mirrors every source file into <vault>\01 Projects\Octavia\Code\*.md
  (one note per file: source-path frontmatter + fenced code block), refreshes
  _Code Index.md, and regenerates the project notes that mirror a repo document
  (Changelog, Roadmap, Face Protocol) so they cannot drift.

  Uses .NET file IO (UTF-8, no BOM) on purpose: PS 5.1 Get-Content/Set-Content
  default to the ANSI code page and silently mangle UTF-8 (the vault's dot/dash/
  bullet glyphs). No-ops gracefully when the vault isn't present on this machine
  so it is safe to run from a hook.

.PARAMETER VaultPath
  Root of the Obsidian vault. Defaults to $env:OCTAVIA_VAULT, else
  'C:\Obsidian Vaults\Vibe Projects'.

.PARAMETER PruneOrphans
  Delete Code\ notes whose source file no longer exists. Off by default (the
  auto hook never prunes); run manually with -PruneOrphans to clean up after a
  source file is renamed or removed.

.EXAMPLE
  powershell -File tools\sync-vault.ps1
.EXAMPLE
  powershell -File tools\sync-vault.ps1 -PruneOrphans
#>
[CmdletBinding()]
param(
  [string]$VaultPath = $(if ($env:OCTAVIA_VAULT) { $env:OCTAVIA_VAULT } else { 'C:\Obsidian Vaults\Vibe Projects' }),
  [switch]$PruneOrphans
)

$ErrorActionPreference = 'Stop'

$repo = Split-Path $PSScriptRoot -Parent          # tools\ -> repo root
$proj = Join-Path $VaultPath '01 Projects\Octavia'
$code = Join-Path $proj 'Code'

if (-not (Test-Path -LiteralPath $VaultPath)) {
  Write-Host "sync-vault: vault not found at '$VaultPath' - skipping (set OCTAVIA_VAULT or pass -VaultPath)."
  exit 0
}
if (-not (Test-Path -LiteralPath $code)) { New-Item -ItemType Directory -Force $code | Out-Null }

$enc = New-Object System.Text.UTF8Encoding($false)   # UTF-8, no BOM

function Get-Lang($ext) {
  switch ($ext) {
    '.cs'       { 'csharp' }
    '.xaml'     { 'xml' }
    '.csproj'   { 'xml' }
    '.manifest' { 'xml' }
    '.config'   { 'xml' }
    '.js'       { 'javascript' }
    '.css'      { 'css' }
    '.html'     { 'html' }
    '.json'     { 'json' }
    '.md'       { 'markdown' }
    '.ps1'      { 'powershell' }
    default     { '' }
  }
}

# --- collect source files ---
# wwwroot\lib is vendored third-party code (three.js, GLTFLoader, three-vrm) — not our
# source, and 3 MB of it. Excluded so the vault holds what we wrote and nothing else.
$files = Get-ChildItem $repo -Recurse -File -Include *.cs,*.xaml,*.csproj,*.manifest,*.js,*.css,*.html,*.ps1,*.config |
  Where-Object {
    $_.FullName -notmatch '\\obj\\' -and
    $_.FullName -notmatch '\\bin\\' -and
    $_.FullName -notmatch '\\dist\\' -and
    $_.FullName -notmatch '\\wwwroot\\lib\\'
  }
foreach ($doc in 'versions.md','README.md','ROADMAP.md','PROTOCOL.md') {
  $p = Join-Path $repo $doc
  if (Test-Path -LiteralPath $p) { $files += Get-Item $p }
}

$expected = @{}
$rows = New-Object System.Collections.Generic.List[string]

foreach ($f in ($files | Sort-Object FullName)) {
  $rel  = $f.FullName.Substring($repo.Length + 1)
  $note = ($rel -replace '\\', ' - ') + '.md'
  $expected[$note] = $true

  $src  = [IO.File]::ReadAllText($f.FullName)
  $open = '```' + (Get-Lang $f.Extension)            # precompute: PS 5.1 comma binds tighter than +
  $body = "---`nproject: Octavia`ntags: [octavia, code]`nsource-path: $rel`n---`n`n# $rel`n`n" +
          $open + "`n" + $src.TrimEnd() + "`n" + '```' + "`n"
  [IO.File]::WriteAllText((Join-Path $code $note), $body, $enc)
  $rows.Add("- [[" + ($note -replace '\.md$', '') + "|" + $rel + "]]")
}

# --- _Code Index.md (live count / date / version) ---
$csproj  = [IO.File]::ReadAllText((Join-Path $repo 'src\Octavia.App\Octavia.App.csproj'))
$version = [regex]::Match($csproj, '<Version>([^<]+)</Version>').Groups[1].Value
$today   = (Get-Date).ToString('d MMMM yyyy')
$index = "---`nproject: Octavia`ntags: [octavia, code]`n---`n`n# Code Snapshot Index`n`n" +
         "Auto-generated by ``tools\sync-vault.ps1`` on **$today** at **v$version** ($($rows.Count) files).`n" +
         "Live repo (source of truth): ``C:\Projects\Octavia`` - not yet on GitHub. Restore procedure: [[Restore From Snapshot]].`n`n" +
         ($rows -join "`n") + "`n"
[IO.File]::WriteAllText((Join-Path $code '_Code Index.md'), $index, $enc)

# --- project notes that mirror a repo document ---
# These were hand-copied until v0.5.0 and drifted every time the repo changed, which is
# the one failure a snapshot tool must not have. Generated now, so they cannot.
$mirrors = @(
  @{ Source = 'versions.md'; Note = 'Changelog.md';     Title = 'Changelog';                   Tag = 'changelog'; Note2 = '(newest-first)' }
  @{ Source = 'ROADMAP.md';  Note = 'Roadmap.md';       Title = 'Roadmap';                     Tag = 'roadmap';   Note2 = '' }
  @{ Source = 'PROTOCOL.md'; Note = 'Face Protocol.md'; Title = 'Octavia Face Protocol';       Tag = 'protocol';  Note2 = '' }
)

foreach ($m in $mirrors) {
  $sourcePath = Join-Path $repo $m.Source
  if (-not (Test-Path -LiteralPath $sourcePath)) { continue }

  # Drop the source's own H1; the note carries its title in the header below.
  $body = ([IO.File]::ReadAllText($sourcePath).TrimEnd() -replace '(?m)\A#\s+[^\r\n]*\r?\n', '').TrimStart()
  $banner = "> Mirror of ``$($m.Source)`` in the repo $($m.Note2)".TrimEnd() +
            ". Auto-generated by ``tools\sync-vault.ps1``."
  $text = "---`nproject: Octavia`ntags: [octavia, $($m.Tag)]`n---`n`n# $($m.Title)`n`n" +
          $banner + "`n`n" + $body + "`n"
  [IO.File]::WriteAllText((Join-Path $proj $m.Note), $text, $enc)
}

# --- orphans (Code notes whose source is gone) ---
$orphans = Get-ChildItem $code -File -Filter *.md |
  Where-Object { $_.Name -ne '_Code Index.md' -and -not $expected.ContainsKey($_.Name) }
foreach ($o in $orphans) {
  if ($PruneOrphans) {
    Remove-Item -LiteralPath $o.FullName -Force
    Write-Host "sync-vault: pruned orphan (source gone): $($o.Name)"
  } else {
    Write-Host "sync-vault: ORPHAN (source gone, run -PruneOrphans to delete): $($o.Name)"
  }
}

Write-Host "sync-vault: wrote $($rows.Count) code notes + _Code Index to '$code', and $($mirrors.Count) mirrored notes."
```
