# tools\sync-vault.ps1

```powershell
<#
.SYNOPSIS
  Regenerate the Obsidian vault's Octavia Android code snapshot from this repo.

.DESCRIPTION
  The sibling of `tools\sync-vault.ps1` in her repo, and deliberately the same shape: one
  note per source file (frontmatter + fenced code), a regenerated `_Code Index`, and the
  project notes that mirror a repo document so they cannot drift.

  The snapshot is the readable, human-diffable copy. GitHub is the primary off-machine one;
  this is what `Restore From Snapshot` is written against and what survives a repo nobody
  can reach.

  Uses .NET file IO (UTF-8, no BOM) on purpose: PS 5.1 Get-Content/Set-Content default to
  the ANSI code page and silently mangle UTF-8 - the dashes and bullets in these notes.
  No-ops gracefully when the vault isn't present, so it is safe to run from a hook.

.PARAMETER VaultPath
  Root of the Obsidian vault. Defaults to $env:OCTAVIA_VAULT, else
  'C:\Obsidian Vaults\Vibe Projects'.

.PARAMETER PruneOrphans
  Delete Code\ notes whose source file no longer exists. Off by default; run manually with
  -PruneOrphans after a rename or a deletion.

.EXAMPLE
  pwsh -NoProfile -ExecutionPolicy Bypass -File tools\sync-vault.ps1
#>
[CmdletBinding()]
param(
  [string]$VaultPath = $(if ($env:OCTAVIA_VAULT) { $env:OCTAVIA_VAULT } else { 'C:\Obsidian Vaults\Vibe Projects' }),
  [switch]$PruneOrphans
)

$ErrorActionPreference = 'Stop'

$repo = Split-Path $PSScriptRoot -Parent          # tools\ -> repo root
$proj = Join-Path $VaultPath '01 Projects\Octavia Android'
$code = Join-Path $proj 'Code'

if (-not (Test-Path -LiteralPath $VaultPath)) {
  Write-Host "sync-vault: vault not found at '$VaultPath' - skipping (set OCTAVIA_VAULT or pass -VaultPath)."
  exit 0
}
if (-not (Test-Path -LiteralPath $code)) { New-Item -ItemType Directory -Force $code | Out-Null }

$enc = New-Object System.Text.UTF8Encoding($false)   # UTF-8, no BOM

function Get-Lang($ext) {
  switch ($ext) {
    '.kt'   { 'kotlin' }
    '.kts'  { 'kotlin' }
    '.xml'  { 'xml' }
    '.toml' { 'toml' }
    '.pro'  { '' }
    '.md'   { 'markdown' }
    '.ps1'  { 'powershell' }
    default { '' }
  }
}

# --- collect source files ---
# build\ and .gradle\ are generated, and the wrapper jar is a binary. What is left is what
# we wrote.
$files = Get-ChildItem $repo -Recurse -File -Include *.kt,*.kts,*.xml,*.toml,*.pro,*.ps1 |
  Where-Object {
    $_.FullName -notmatch '\\build\\' -and
    $_.FullName -notmatch '\\\.gradle\\' -and
    $_.FullName -notmatch '\\\.idea\\'
  }
foreach ($doc in 'versions.md','README.md','ROADMAP.md') {
  $p = Join-Path $repo $doc
  if (Test-Path -LiteralPath $p) { $files += Get-Item $p }
}

$expected = @{}
$rows = New-Object System.Collections.Generic.List[string]

foreach ($f in ($files | Sort-Object FullName)) {
  $rel  = $f.FullName.Substring($repo.Length + 1)
  $note = ($rel -replace '\\', ' - ') + '.md'
  $expected[$note] = $true

  $src  = [IO.File]::ReadAllText($f.FullName)
  $open = '```' + (Get-Lang $f.Extension)            # precompute: PS 5.1 comma binds tighter than +
  $body = "---`nproject: Octavia Android`ntags: [octavia, octavia-android, code]`nsource-path: $rel`n---`n`n# $rel`n`n" +
          $open + "`n" + $src.TrimEnd() + "`n" + '```' + "`n"
  [IO.File]::WriteAllText((Join-Path $code $note), $body, $enc)
  $rows.Add("- [[" + ($note -replace '\.md$', '') + "|" + $rel + "]]")
}

# --- _Code Index.md ---
# The version lives in versions.md rather than a project file, so it is read from the
# newest header. One source for the number, the same way her csproj is for hers.
$versionsText = [IO.File]::ReadAllText((Join-Path $repo 'versions.md'))
$version = [regex]::Match($versionsText, '(?m)^##\s+(\d+\.\d+\.\d+)').Groups[1].Value
$today   = (Get-Date).ToString('d MMMM yyyy')
$index = "---`nproject: Octavia Android`ntags: [octavia, octavia-android, code]`n---`n`n# Code Snapshot Index`n`n" +
         "Auto-generated by ``tools\sync-vault.ps1`` on **$today** at **v$version** ($($rows.Count) files).`n" +
         "Live repo (source of truth): ``C:\Projects\Octavia-Android``, private on GitHub at N3cTr0/Octavia-Android since 08/31/2026.`n`n" +
         ($rows -join "`n") + "`n"
[IO.File]::WriteAllText((Join-Path $code '_Code Index.md'), $index, $enc)

# --- project notes that mirror a repo document ---
$mirrors = @(
  @{ Source = 'versions.md'; Note = 'Changelog.md'; Title = 'Changelog'; Tag = 'changelog'; Note2 = '(newest-first)' }
  @{ Source = 'ROADMAP.md';  Note = 'Roadmap.md';   Title = 'Roadmap';   Tag = 'roadmap';   Note2 = '' }
  @{ Source = 'README.md';   Note = 'Overview.md';  Title = 'Overview';  Tag = 'overview';  Note2 = '' }
)

foreach ($m in $mirrors) {
  $sourcePath = Join-Path $repo $m.Source
  if (-not (Test-Path -LiteralPath $sourcePath)) { continue }

  # Drop the source's own H1; the note carries its title in the header below.
  $body = ([IO.File]::ReadAllText($sourcePath).TrimEnd() -replace '(?m)\A#\s+[^\r\n]*\r?\n', '').TrimStart()
  $banner = "> Mirror of ``$($m.Source)`` in the repo $($m.Note2)".TrimEnd() +
            ". Auto-generated by ``tools\sync-vault.ps1``."
  $text = "---`nproject: Octavia Android`ntags: [octavia, octavia-android, $($m.Tag)]`n---`n`n# $($m.Title)`n`n" +
          $banner + "`n`n" + $body + "`n"
  [IO.File]::WriteAllText((Join-Path $proj $m.Note), $text, $enc)
}

# --- orphans ---
$removed = 0
if ($PruneOrphans) {
  foreach ($note in (Get-ChildItem $code -File -Filter *.md)) {
    if ($note.Name -eq '_Code Index.md') { continue }
    if (-not $expected.ContainsKey($note.Name)) { Remove-Item -LiteralPath $note.FullName -Force; $removed++ }
  }
}

$tail = if ($PruneOrphans) { ", and pruned $removed orphan(s)" } else { '' }
Write-Host "sync-vault: wrote $($rows.Count) code notes + _Code Index to '$code', and $($mirrors.Count) mirrored notes$tail."
```
